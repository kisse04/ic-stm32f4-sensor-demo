/** Embedded API.
  *
  * This is an API for embedded use cases with support for customized storages
  * through a storage interface (pb_embedded_storageI.h) that allows for
  * full memory optimization in environments with limited resources. It contains
  * all functionality needed for a complete fingerprint system, including
  * enrollment and verification.
  *
  * For an example of how to use this API, see pb_embedded_example.c.
  */

/** The embedded object. */
typedef struct pb_embedded_st pb_embedded_t;

/** Specific settings for the embedded API. */
typedef struct pb_embedded_settings_st {
    /** Currently no specific settings, reserved for future use. */
    uint8_t rfu;
} pb_embedded_settings_t;

/**
 * Returns the default embedded settings.
 */
pb_embedded_settings_t
pb_embedded_get_default_settings (void);

/**
 * Creates an embedded object.
 *
 * @param[in] settings the configuration of the embedded object
 * @param[in] algorithm the algorithm used during enrollment and verification
 * @param[out] embedded is the pb_embedded_t object created
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_create (const pb_embedded_settings_t* settings,
                    pb_algorithm_t* algorithm,
                    pb_embedded_t** embedded);

/**
 * Retains an embedded object, which means that the reference counter for the object
 * will increase by 1. The caller must make sure that _delete is being called
 * when the object is not needed anymore.
 *
 * Retaining a null pointer has no effect.
 * @param[in] embedded is the pb_embedded_t object to be retained
 *
 * @return retained pb_embedded_t object
 */
pb_embedded_t*
pb_embedded_retain (pb_embedded_t* embedded);

/**
* Deletes an embedded object.
*
* @param[in] embedded is the pb_embedded_t object deleted
*/

pb_embedded_delete (pb_embedded_t* embedded);

/**
 * Encodes an embedded object.
 *
 * @param[in] embedded is the pb_embedded_t object.
 * @param[out] data is the encoded data. The caller is responsible for
 *             deallocating the returned data by calling pb_embedded_encode_data_delete.
 * @param[out] data_size is the encoded data size.
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_encode (pb_embedded_t* embedded,
                    uint8_t** data,
                    uint32_t* data_size);

/**
 * Decodes an embedded object
 *
 * @param[in] data is the encoded data
 * @param[in] data_size is the encoded data size
 * @param[in] algorithm is the algorithm to use for this embedded object
 * @param[out] embedded is the decoded pb_embedded_t object
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_decode (const uint8_t* data,
                    uint32_t data_size,
                    pb_algorithm_t* algorithm,
                    pb_embedded_t** embedded);

/**
 * Computes the image size to be used for extraction. Call this function to

pb_embedded_scale_image_for_extraction (const pb_embedded_t* embedded,

                                        pb_embedded_image_t* extraction_image);

/**
 * Static Pattern Detection (SPD).
 *
 * Images may contain patterns that is repeated over a number of consecutive images. These
 * patterns may pose a threat if they are enrolled or if they are being updated with into
 * the enrolled template, causing false accepts.
 *
 * The static pattern detection works by comparing the current image against a number of
 * the previous images and if certain areas in the image is similar enough to the same areas
 * in the previous images they are classified as containing static patterns. Those
 * areas would then be masked out allowing the matcher to ignore them and hence avoid false
 * accepts.
 *
 * To be able to compare against previous images, specific features from each image needs
 * to be stored between calls to the static pattern detection. The SPD also needs to store
 * some additional data for optimal detection. This combined data is updated for each call
 * to pb_embedded_spd_init/detect and needs to be stored by the caller using the specified
 * pb_spd_embedded_encode and pb_spd_embedded_decode functions.
 *
 * Note: The Cardo matcher also has a built-in protection against certain types of static
 * patterns (non-symmetric), improving the protection further.
 */

/** The SPD object. */
typedef struct pb_spd_st pb_spd_t;

/* Settings for the SPD. */
typedef struct pb_embedded_spd_settings_st {
    /** The sensitivity for Static Pattern Detection (SPD) with values from -100 to 100.
      * A negative value makes the SPD less sensitive to static patterns and thus detects
      * less static patterns while a positive value makes the SPD more sensitive and
      * detecting more static pattern. The sensitivity will affect the internal threshold
      * (tuned for the specific algorithm). Default value is 0, meaning that the internal
      * threshold is unaffected. */
    int8_t sensitivity;
} pb_embedded_spd_settings_t;

/** Retrieves default SPD settings. */
pb_embedded_spd_settings_t
pb_embedded_spd_get_default_settings ();

/* Initializes the SPD. Will write the initial SPD data to the storage.
 * This function only needs to be called once at the very start of the
 * system's lifetime. It may be called multiple times to re-init the
 * SPD but then any history of detected static patterns will be lost
 * which may pose a security risk.
 *
 * @param[in] embedded is the embedded object to use.
 * @param[in] extraction_rows is number of rows as returned by
 *            pb_embedded_compute_extraction_image_size.
 * @param[in] extraction_cols is number of columns as returned by
 *            pb_embedded_compute_extraction_image_size.
 * @param[out] spd is the returned spd object. The caller is responsible
 *            for deleting this object (pb_embedded_spd_delete).
 */
pb_rc_t
pb_embedded_spd_init (pb_embedded_t* embedded,
                      uint16_t extraction_rows, uint16_t extraction_cols,
                      pb_spd_t** spd);

/**
 * Detects static patterns in the image. Must be called for every image
 * captured before calling pb_embedded_extract to avoid the risk of matching
 * on static patterns in the image, causing false accepts.
 * This function will allocate and set the image mask in the image with
 * information on which parts of the image that was detected as static patterns.
 *
 * This function also updates the SPD object with new data that needs to be
 * stored between consecutive calls to _detect. If needed, use _encode and
 * _decode to store the SPD object data to flash in between calls.
 *
 * Please note that the added mask will increase the memory needed for extraction
 * by extraction_rows * extraction_cols byte.
 *
 * @param[in] embedded is the embedded object to use.
 * @param[in/out] extraction image is the extraction image in which to detect
 *            static patterns. Will be updated with a mask of where static
 *            pattern was detected. Please note that the mask must be
 *            deallocated by the caller.
 * @param[in] spd_settings is the settings for the static pattern detection.
 * @param[in/out] spd is the SPD object. The SPD object is being updated for each
 *            call to _detect with new information of the current image to be
 *            used by later calls to _detect.
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_spd_detect (pb_embedded_t* embedded,
                        pb_embedded_image_t* extraction_image,
                        const pb_embedded_spd_settings_t* spd_settings,
                        pb_spd_t* spd);

/** Encodes the SPD object data.
  *
  * @param[in] spd is the SPD object.
  * @param[out] data is the returned encoded data. The caller is responsible for
  *             deallocating the returned data by calling pb_embedded_encode_data_delete.
  * @param[out] data_size is the size of the returned data. The size of the SPD object
  *             data depends on the image size and which algorithm that is used. After
  *             approximately 5-10 calls to pb_embedded_spd_detect the SPD object data
  *             has reached it's maximum size of about 5-15kB.
  *
  * @return PB_RC_OK if successful, otherwise an error code.
  */
pb_rc_t
pb_embedded_spd_encode (pb_spd_t* spd,
                        uint8_t** data,
                        uint32_t* data_size);

/** Decodes the SPD data into a SPD object. */
pb_rc_t
pb_embedded_spd_decode (const uint8_t* data,
                        uint32_t data_size,
                        pb_spd_t** spd);

/** Deletes the SPD object. */
void
pb_embedded_spd_delete (pb_spd_t* spd);

/**
 * Latent Detection (LD).
 *
 * The latent detection (LD) module is used to detect if the current image is from a
 * real new fingerprint or if it is an image of residues of previous fingerprint left
 * on sensor, a latent fingerprint. Latent fingerprints are mostly a threat to optical
 * sensors.
 *
 * When a verification is done and positive match result is received, the LD module
 * may be called to check if the current image is latent.
 *
 * The latent data must be stored between verifications, since latent detection uses
 * information from previous verifications. Use pb_embedded_latent_encode/decode to
 * serialize data between verifications if needed.
 *
 * Please note that the LD module can only be used with algorithms that support latent
 * detection, i.e. for algorithms that has been trained for latent detection.
 */

/** The LD object. */
typedef struct pb_latent_st pb_latent_t;

/* Settings for the LD. */
typedef struct pb_embedded_latent_settings_st {
    /** The FRR penalty for LD. Running latent detection may also falsely reject real
      * fingerprints as being latent. This setting determines the amount of false rejects
      * that can be expected from the LD. A higher FRR penalty will result in more latent
      * fingerprints being rejected and vice versa.
      * Default value is 0.5% (PB_FRR_0_5). */
    pb_frr_t frr_penalty;
} pb_embedded_latent_settings_t;

/** Retrieves default LD settings. */
pb_embedded_latent_settings_t
pb_embedded_latent_get_default_settings ();

/* Initializes the LD object. This function only needs to be called once at the very start
 * of the system's lifetime.
 *
 * @param[in] embedded is the embedded object to use.
 * @param[out] latent is the returned latent object. The caller is responsible
 *            for deleting this object (pb_embedded_latent_delete).
 */
pb_rc_t
pb_embedded_latent_init (pb_embedded_t* embedded,
                         pb_latent_t** latent);

/*
 Initializes the LD object. This function is only for testing purpose 
 */
pb_rc_t
pb_embedded_latent_init_ext (pb_embedded_t* embedded, pb_latent_t** latent_out,
                             pb_template_t* T1, pb_template_t* T2,int update);

/**
 * Detects if current verification image is from a real or latent fingerprint. This function shall
 * be called after each accepted verification. The match result will be updated to reflect the
 * result from the latent detection.
 *
 * @param[in] embedded is the embedded object to use.
 * @param[in] verification_image is the current verification image.
 * @param[in] latent_settings is the settings for the latent detection.
 * @param[in/out] latent is the LD object. The LD object is being updated for each call to _detect
 *            with new information of the current image to be used by later calls to _detect.
 * @param[in/out] match_result is the match result object returned by the call to pb_embedded_verify.
 *            Calling pb_embedded_latent_detect will update the decision of the match result,
 *            use pb_match_result_get_decision to get the final decision for the match inlcuding
 *            both matcher and LD decisions. Additional details of the latent attack detection
 *            may be retrieved by calling pb_match_result_is_latent.
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_latent_detect (pb_embedded_t* embedded,
                           const pb_embedded_image_t* verification_image,
                           const pb_embedded_latent_settings_t* latent_settings,
                           pb_latent_t* latent,
                           pb_match_result_t* match_result);

/** Encodes the LD object data.
  *
  * @param[in] latent is the LD object.
  * @param[out] data is the returned encoded data. The caller is responsible for
  *             deallocating the returned data by calling pb_embedded_encode_data_delete.
  * @param[out] data_size is the size of the returned data. The size of the LD object
  *             data depends on the image size and which algorithm that is used.
  *
  * @return PB_RC_OK if successful, otherwise an error code.
  */
pb_rc_t
pb_embedded_latent_encode (const pb_latent_t* latent,
                           uint8_t** out_data,
                           uint32_t* out_data_size);

/** Decodes the LD data into a LD object. */
pb_rc_t
pb_embedded_latent_decode (const uint8_t* data,
                           uint32_t data_size,
                           pb_latent_t** latent);

/** Deletes the LD object. */
void
pb_embedded_latent_delete (pb_latent_t* latent);

/** Deletes the encode data returned from pb_embedded_encode, pb_embedded_spd_encode,
  * or pb_embedded_latent_encode. */
void
pb_embedded_encode_data_delete (uint8_t* data);

/** Enrollment process.
  *
  * The below example describes on a high level what needs to be done to enroll a finger
  * into a storage. Error handling has been left out in order to simplify the example.
  *
  *     // Setup storage. The caller has full control over how and where the enrolled templates
  *     // are stored and how they are accessed.
  *     storage_context = external_storage_create (...);
  *
  *     // Begin enrollment.
  *     pb_embedded_enroll_begin (embedded, storage_implementation, storage_context);
  *
  *     // Enroll x templates.
  *     do {
  *         // Capture an image.
  *         image = external_capture_image (...);
  *
  *         // Extract a subtemplate to be enrolled from the captured image.
  *         pb_embedded_extract (embedded, image, &template);
  *
  *         // (Optionally) Screen the extracted template.
  *         pb_embedded_extract_screen (embedded, image, &quality);
  *         if (!external_quality_is_ok (&quality)) continue;
  *         pb_embedded_enroll_screen (embedded, template, screening_settings, &geometry);
  *         if (!external_geometry_is_ok (&geometry)) continue;
  *
  *         // Add template to enrollment.
  *         pb_embedded_enroll_add (embedded, template);
  *
  *     } while (! enrollment_is_done);
  *
  *     // End enrollment.
  *     pb_embedded_enroll_end (embedded);
  *
  * During the call to pb_embedded_enroll_add the _write_subtemplate function of the specified
  * storage interface will be called to allow the implementation to determine if, how and where the
  * subtemplate is stored.
  */

/**
 * Begin enrollment of a finger. The caller decides the storage interface
 * to be used by the enrollment process and which finger to enroll.
 *
 * The storage interface and context is stored in the embedded object until
 * pb_embedded_enroll_end is called.
 *
 * @param[in] embedded is the pb_embedded_t object to use.
 * @param[in] storage_interface is the storage interface that will determine
 *            where and how the enrolled subtemplates will be stored.
 * @param[in] storage_context is the context for the storage interface.
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_enroll_begin (pb_embedded_t* embedded,
                          const pb_embedded_storageI* storage_interface,
                          void* storage_context);

typedef struct {
    /** The maximum overlap against any of the previously added images,
      * in percentage of the total image area. A maximum overlap of 90
      * or lower is recommended to ensure a good spread of the
      * enrollment images.
      * Note that this value will be affected by the value of the
      * nbr_of_templates_to_check_redundancy_against setting. */
    uint8_t maximum_overlap;
    /** The overlap against the last added image. An overlap of 80 or
      * lower is recommended to ensure a good spread of the
      * enrollment images. */
    uint8_t overlap_against_last_added_image;
    /** The index of the previously enrolled finger (storage context) that
      * the enrollment template matched against or -1 if it didn't match
      * any previously enrolled fingers. */
    int matched_against_finger;
} pb_embedded_enroll_geometry_t;

typedef struct pb_embedded_enroll_screen_settings_st {
    /** The number of templates to check if the current template is very similar too. If the
      * template is similar then it contains redundant information and may be rejected for enrollment.
      * This setting can be used to limit the number of templates to check redundancy against or even
      * disable it completely (0). Default 1 (only check against last added template). */
    int nbr_of_templates_to_check_redundancy_against;

    /** List of contexts for the storage interface, each corresponding to one stored finger, that the
      * enrollment template will be verified against to determine if the same finger has already been
      * enrolled or not. If the enrollment template matches any previously enrolled finger the
      * matched_against_finger of the pb_embedded_enroll_geometry_t will be set accordingly. If
      * multiple enrollments of the same finger is allowed, just set this to NULL. Default NULL. */
    void** storage_contexts;
    /** The number of stored fingers to verify the enrolled template against. Default 0. */
    int nbr_of_fingers;
} pb_embedded_enroll_screen_settings_t;

/**
 * Returns the default screening settings for enroll.
 */
pb_embedded_enroll_screen_settings_t
pb_embedded_get_default_enroll_screen_settings (void);

/**
 * Template screening before adding template into enrollment.
 *
 * Note: This is an recommended but optional step
 *
 * @param[in] embedded is the pb_embedded_t object to use.
 * @param[in] enroll_template is the template to screen for enrollment.
 * @param[in] screening_settings are the settings to be used in the
 *            screening process.
 * @param[out] geometry is the result of the enrollment screening.
 *
 * @return PB_RC_OK if successful, otherwise an error code.

pb_rc_t
pb_embedded_enroll_screen (pb_embedded_t* embedded,
                           pb_template_t* enroll_template,

/**

pb_rc_t

/** Verification process.
  *
  * The below example describes on a high level what needs to be done to verify an image against
  * a number of enrolled fingers. Error handling has been left out in order to simplify the example.
  *
  *     // Retrive stored fingers.
  *     storage_contexts = external_storage_get_fingers (...);
  *     nbr_of_fingers = external_storage_get_nbr_of_fingers (...);
  *
  *     // Capture an image.
  *     image = external_capture_image (...);
  *
  *     // Extract a template from the captured image.
  *     pb_embedded_extract (embedded, image, &template);
  *
  *     // Verify.
  *     pb_embedded_verify (embedded, storage_implementation, storage_contexts, nbr_of_fingers,
  *                         verify_settings, template, PB_FAR_50K, &match_result);
  *
  *     // Check decision.
  *     if (pb_match_result_get_decision (match_result) == PB_DECISION_MATCH) {
  *         // Allow access!
  *     }
  *
  *     // Finalize verification.
  *     pb_embedded_verify_finalize (embedded);
  *
  * During the call to pb_embedded_verify_finalize, in the case of an accept, the _write_subtemplate
  * function of the specified storage interface will be called to allow the implementation to
  * determine if the enrolled templates shall be updated with the verification template, either just
  * adding it or replacing any previously added subtemplate.
  */

typedef struct pb_embedded_verify_settings_st {
    /**
     * Adjusts the latency for the verification. Allows for a decrease or increase of
     * the latency for a verification in relation to the default latency. The latency
     * adjustment is given as a percentage of the default latency, with negative values
     * indicating a decrease of latency, with -100 as the minimum value allowed.
     * Default value is 0, meaning no increase or decrease of latency.
     *
     * Adjusting the latency affects the biometric performance, the shorter the
     * verification time, the worse FRR.
     *
     * The default latency is dependent on how the algorithm has been setup. A change of
     * algorithm settings may therefore require a change of the latency adjustments.
     *
     * Note that this setting only affects the verification part, excluding such things
     * as static pattern detection, image processing and feature extraction.
     */
    int latency_adjustment;

    /**
     * Tells if dynamic latency adjustment shall be used or not. Dynamic latency adjustment
     * applies latency adjustment 100%, but only for specific cases, allowing for an
     * improvement in FRR without affecting latency for other cases. Dynamic latency
     * adjustment is only applied in approximately 1% of all verifications, for both
     * genuines and impostors.
     *
     * Note that specifically setting latency adjustment to 100% will result in even better
     * FRR but then with a latency hit for all verifications. The exact difference in
     * performance depends on what condition and dataset that is used.
     *
     * Also note that specifically setting latency adjustment to 100% or higher will
     * effectively disable dynamic latency adjustment.
     *
     * Default value is 0, meaning that dynamic latency adjustment is disabled by default.
     */
    int dynamic_latency_adjustment;

    /**
      * Runs the verification in batches to reduce memory footprint, especially beneficial
      * if execute in place (XIP) can't be used for the stored templates. The batch size
      * specifies the number of fingers to run for each batch. Default value is 0, meaning
      * that all fingers are run in one single batch.
      *
      * When using this feature it is recommended to supply the list of contexts sorted
      * by how likely it is that each finger will be the finger currently being verified,
      * e.g. by sorting in the order of when a finger was most recently authenticated.
      *
      * The latency of a batch verification will be similar to non-batch verification for
      * a reject, but may be both shorter or longer for an accept depending on if the
      * verification finger is in the first batch or not, respectively.
      */
    int batch_size;

    /** Enables automatic adjustment of FAR level depending on the number of enrolled fingers that
      * is verified against. Default value is 0 (disabled), meaning that the caller has to adjust the
      * requested FAR depending on the number of enrolled fingers that is verified against.
      * E.g. to achieve an actual FAR of 1/10K when verifiying against 5 enrolled fingers the caller
      * needs to set the requested FAR to 1/50K (5 * 1/50K = 1/10K) if this option is disabled. If
      * enabled the caller can set 1/10K independently of the number of enrolled fingers that is
      * verified against. */
    int automatic_adjustment_of_far;

    /**
      * Sets the minimum match area to be required for an accept, in percentage of the total image
      * area. Default 0, which means that no requirement is put on match area. This setting may be
      * used to further increase the security of the system e.g. for payment use cases or if there
      * are indications that an impostor is trying to attack the system. This setting has no effect
      * on the requested FAR but it is recommended to also lower the requested FAR when this setting
      * is enabled.
      * The match area is defined as the parts of the verification image that matched the enrolled image,
      * excluding both fingerprint areas that did not match as well as non-fingerprint areas (background).
      */
    int minimum_match_area;

    /** Sets after how long time the verification shall be aborted, in percentage (0-100) of the total
      * verification progress. This may be used instead of implementing the run_interleaved function
      * to make sure a verification does not take too long time. A verification will return
      * immediately when a match has been found which means that most genuine verifications will be
      * performed quickly, but in some genuine cases and for impostors the verification will take
      * longer time and then this feature may be used to set a maximum limit.
      * Default value is 0, which means that no early abort will be performed. */
    int abort_verification_at;
} pb_embedded_verify_settings_t;

/**
 * Returns the default verification settings.
 */
pb_embedded_verify_settings_t
pb_embedded_get_default_verification_settings (void);

/**
 * Verifies a template against a list of previously enrolled fingers.
 *
 * The caller must assure that pb_embedded_verify_finalize is called after
 * a succesful call to this function, independent of whether the verification
 * template was accepted or rejected.
 *
 * The storage interface and contexts are stored in the embedded object until
 * pb_embedded_verify_finalize is called.
 *
 * @param[in] embedded is the pb_embedded_t object.
 * @param[in] storage_interface is the storage interface that determines where
 *            the enrolled fingers are stored and how to access them.
 * @param[in] storage_contexts is the list of contexts for the storage interface,
 *            each corresponding to one stored finger.
 * @param[in] nbr_of_fingers is the number of fingers to verify against.
 * @param[in] verification_template is the template to use for verification.
 * @param[in] false_accept_rate is the requested false accept rate if verifying against
 *            one single enrolled template. Note that the false accept rate needs to be
 *            adjusted with respect to the number of enrolled templates, e.g. to achieve
 *            an actual FAR of 1/10K when verifying against 5 fingers, a FAR level of
 *            1/50K must be set (5 * 1/50K = 1/10K). The caller may also use the verify
 *            setting automatic_adjustment_of_far to automatically adjust the FAR
 *            depending on the number of enrolled templates that is verified against.
 * @param[out] match_result is the returned result of the verification.
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_verify (pb_embedded_t* embedded,
                    const pb_embedded_storageI* storage_interface,
                    void** storage_contexts,
                    uint8_t nbr_of_fingers,
                    const pb_embedded_verify_settings_t* verify_settings,
                    const pb_template_t* verification_template,
                    pb_far_t false_accept_rate,
                    pb_match_result_t** match_result);

/**
 * After a verification has completed without error, this function will finalize any remaining
 * work, such as updating an enrolled template, in which case the write_subtemplate function
 * of the storage interface will be called. The implementation of the storage interface may
 * then decide if the verification template shall be used for update or not.
 *
 * This function must be called regardless of the match result from the pb_embedded_verify call.
 *
 * @param[in] embedded is the pb_embedded_t object.
 *
 * @return PB_RC_OK if successful, otherwise an error code.
 */
pb_rc_t
pb_embedded_verify_finalize (pb_embedded_t* embedded);

/** Returns the supported image size and resolution. */
pb_rc_t
pb_embedded_get_supported_image_size (const pb_embedded_t* embedded,
                                      uint16_t* rows, uint16_t* cols,
                                      uint16_t* resolution);
