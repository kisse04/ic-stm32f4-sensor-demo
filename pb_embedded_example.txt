 *

 * using the Precise Embedded API (pb_embedded.h).
 *

 * use the appropriate library.
 *
 * gcc Examples/pb_embedded_example.c Examples/pb_embedded_example_external.c -IBMF/inc -LBMF/lib/x86_64-linux-gnu/ -lBMF -lm -o pb_embedded_example && ./pb_embedded_example
 *

/** The maximum number of enrollment captures. */
#define PB_EMBEDDED_EXAMPLE_MAX_ENROLLMENT_CAPTURES 16

/** The false accept rate for a verification. */
#define PB_EMBEDDED_EXAMPLE_FAR PB_FAR_50K

/** The number of fingers to enroll. */
#define PB_EMBEDDED_EXAMPLE_NBR_OF_FINGERS_TO_ENROLL 4

/** The number of verifications. */
#define PB_EMBEDDED_EXAMPLE_NBR_OF_VERIFICATIONS  50

/** The maximum number of fingers to verify against at the same time.
 *
 *  Instead of matching againt all enrolled fingers at once, it can be done in batches.
 *  For example if you do batches of 5 fingers at a time you can skip the remaining batches
 *  as soon as you find a match. This will save time if the match can be found in
 *  an early batch.
 *  We therefore keep a history of matched fingers and first try to match with the
 *  fingers that we most recently matched against, as these are more likley to be the ones
 *  accessing the system again.
 *
 *  The heap memory peak will also decrease as not all subtemplates will be read from storage at
 *  the same time.
 *
 *  Note: Using several smaller batches instead of a single big batch could increase latency if
 *        the latest matched fingers does not have a higher probability of accessing the system again.
 *
 *        Ex1: If only a few fingers (when many are enrolled) often access the system,
 *             then smaller batches are better.
 *
 *        Ex2: If all fingers are almost equally likley to access the system,
 *             then a single big batch is better.
 *             (e.g. office door access were all employees only enter 1-2 times a day)
*/
#define PB_EMBEDDED_EXAMPLE_MAX_FINGER_BATCH_SIZE   10

#define CHECK_STATUS(status, error_text) \
do { \
    if (status != PB_RC_OK) { \
        printf (error_text, status); \
        return status; \
    } \
} while (0) \

static pb_rc_t
create_embedded (pb_embedded_t** embedded)
{
    /* Get algorithm. */
    const pb_algorithmI* algorithm_interface = external_get_algorithm ();
    if (algorithm_interface == NULL) return PB_RC_FATAL;

    /* Create algorithm. */
    pb_session_t* session = pb_session_create ();
    pb_algorithm_t* algorithm = algorithm_interface->create (session);
    pb_session_delete (session);
    session = NULL;
    if (algorithm == NULL) return PB_RC_FATAL;

    /* Create embedded object with default settings. */
    pb_embedded_settings_t embedded_settings = pb_embedded_get_default_settings ();
    pb_rc_t status = pb_embedded_create (&embedded_settings, algorithm, embedded);
    pb_algorithm_delete (algorithm);
    algorithm = NULL;

    return status;
}

/** Flash interface. */

static void* FlashContext = NULL;
/** Flash subtemplate storage. */
static pb_flash_storage_t* FlashSubtemplateStorage = NULL;
/** Flash SPD storage. */
static pb_flash_data_storage_t* FlashSPDStorage = NULL;

#ifdef PB_EMBEDDED_EXAMPLE_USE_LATENT
/** Flash Latent storage. */
static pb_flash_data_storage_t* FlashLatentStorage = NULL;

#define PB_EMBEDDED_EXAMPLE_LATENT_FRR_PENALTY PB_FRR_0_5

#endif // PB_EMBEDDED_EXAMPLE_USE_LATENT

static pb_rc_t
save_spd (pb_spd_t* spd)
{
    pb_rc_t status = PB_RC_OK;
    uint8_t* data = NULL;
    uint32_t data_size = 0;

    /* Encode embedded object. */
    status = pb_embedded_spd_encode (spd, &data, &data_size);
    CHECK_STATUS (status, "Encoding SPD object failed with error %d\n");
    /* Save data. */
    status = pb_flash_data_storage_write (FlashSPDStorage, data, data_size);
    pb_embedded_encode_data_delete (data);
    data = NULL;
    CHECK_STATUS (status, "Saving SPD data failed with error %d\n");

    return status;
}

static pb_rc_t
load_spd (pb_spd_t** spd)
{
    pb_rc_t status = PB_RC_OK;

    /* Load SPD data. */
    uint8_t* data = NULL;
    uint32_t data_size = 0;
    status = pb_flash_data_storage_read (FlashSPDStorage, &data, &data_size);
    CHECK_STATUS (status, "Loading SPD data failed with error %d\n");

    /* Decode embedded data into embedded object. */
    status = pb_embedded_spd_decode (data, data_size, spd);
    free (data);
    CHECK_STATUS (status, "Decoding SPD object failed with error %d\n");

    return status;
}

static pb_rc_t
pb_latent_detection (pb_embedded_t *embedded, pb_embedded_image_t *latent_image, pb_match_result_t* match_result)
{
    pb_rc_t status = PB_RC_OK;
    pb_latent_t* latent_detect= NULL;

    /* Load latent detect object. */
    status = load_latent (&latent_detect);
    if (status != PB_RC_OK) return status;

    pb_embedded_latent_settings_t latent_settings = pb_embedded_latent_get_default_settings ();
    latent_settings.frr_penalty = PB_EMBEDDED_EXAMPLE_LATENT_FRR_PENALTY;

    /* Do Latent detection. The match_result object will be updated with the result. */
    status = pb_embedded_latent_detect (embedded, latent_image,
				&latent_settings, latent_detect, match_result);
    CHECK_STATUS (status, "Latent detection failed with error %d\n");

    /* Save the latent object. */
    save_latent (latent_detect);
    if (status != PB_RC_OK) return status;

    pb_embedded_latent_delete (latent_detect);

    return status;
}
#endif // PB_EMBEDDED_EXAMPLE_USE_LATENT

static pb_rc_t
init ()
{
    pb_rc_t status = PB_RC_OK;

    /* Create embedded object. */
    pb_embedded_t* embedded = NULL;
    status = create_embedded (&embedded);
    CHECK_STATUS (status, "Creating embedded object failed with error %d\n");

    /* Compute extraction image size (for input to SPD initialization). */
    uint16_t extraction_rows = 0;
    uint16_t extraction_cols = 0;
    status = pb_embedded_compute_extraction_image_size (embedded, IMAGE_ROWS, IMAGE_COLS, IMAGE_RESOLUTION, &extraction_rows, &extraction_cols);
    CHECK_STATUS (status, "Computing extraction size failed with error %d\n");

    /* Initialize SPD. */
    pb_spd_t* spd = NULL;
    status = pb_embedded_spd_init (embedded, extraction_rows, extraction_cols, &spd);
    CHECK_STATUS (status, "Static pattern initialization failed with error %d\n");

#ifdef PB_EMBEDDED_EXAMPLE_USE_LATENT
    /* Initialize Latent. */
    pb_latent_t* latent = NULL;
    status = pb_embedded_latent_init (embedded, &latent);
    CHECK_STATUS (status, "Latent initialization failed with error %d\n");
#endif // PB_EMBEDDED_EXAMPLE_USE_LATENT

    /* Delete embedded object. */
    pb_embedded_delete (embedded);
    embedded = NULL;

    /* Initialize flash implementation. */
    status = external_flash_init (&FlashI, &FlashContext);
    CHECK_STATUS (status, "Initializing flash implementation failed with error %d\n");

    /* Initialize flash subtemplate storage. */
    pb_flash_storage_settings_t settings = pb_flash_storage_get_default_settings ();
    settings.start_address = EXTERNAL_SUBTEMPLATE_STORAGE_START_ADDRESS;
    settings.end_address = EXTERNAL_SUBTEMPLATE_STORAGE_END_ADDRESS;
    settings.max_subtemplate_size = EXTERNAL_MAX_SUBTEMPLATE_SIZE;
    status = pb_flash_storage_init (&settings, FlashI, FlashContext, &FlashSubtemplateStorage);
    CHECK_STATUS (status, "Initializing flash subtemplate storage failed with error %d\n");

    /* Check how many subtemplates that will be possible to store. */
    const int max_nbr_of_subtemplates = pb_flash_storage_compute_max_nbr_of_subtemplates (FlashSubtemplateStorage);
    printf ("Created a subtemplate storage with room for %d subtemplates.\n", max_nbr_of_subtemplates);

    /* Initialize flash SPD storage. */
    pb_flash_data_storage_settings_t spd_settings = pb_flash_data_storage_get_default_settings ();
    spd_settings.start_address = EXTERNAL_SPD_STORAGE_START_ADDRESS;
    spd_settings.end_address = EXTERNAL_SPD_STORAGE_END_ADDRESS;
    spd_settings.max_object_size = EXTERNAL_MAX_SPD_SIZE;
    status = pb_flash_data_storage_init (&spd_settings, FlashI, FlashContext, &FlashSPDStorage);
    CHECK_STATUS (status, "Initializing flash SPD storage failed with error %d\n");

    /* Encode and save the SPD object. */
    status = save_spd (spd);
    CHECK_STATUS (status, "Static pattern save failed with error %d\n");
    pb_embedded_spd_delete (spd);
    spd = NULL;

#ifdef PB_EMBEDDED_EXAMPLE_USE_LATENT
    /* Initialize flash Latent storage. */
    pb_flash_data_storage_settings_t latent_settings = pb_flash_data_storage_get_default_settings ();
    latent_settings.start_address = EXTERNAL_LATENT_STORAGE_START_ADDRESS;
    latent_settings.end_address = EXTERNAL_LATENT_STORAGE_END_ADDRESS;
    latent_settings.max_object_size = EXTERNAL_MAX_LATENT_SIZE;
    status = pb_flash_data_storage_init (&latent_settings, FlashI, FlashContext, &FlashLatentStorage);
    CHECK_STATUS (status, "Initializing flash Latent storage failed with error %d\n");

    /* Encode and save the Latent object. */
    status = save_latent (latent);
    CHECK_STATUS (status, "Static pattern save failed with error %d\n");
    pb_embedded_latent_delete (latent);
    latent = NULL;
#endif // PB_EMBEDDED_EXAMPLE_USE_LATENT

    return status;
}

static pb_rc_t
capture_image (pb_embedded_t* embedded, 
               pb_embedded_image_t** image)
{
    pb_rc_t status = PB_RC_OK;

    /* Capture an image from the sensor. */
    status = external_sensor_capture_image (image);
    CHECK_STATUS (status, "Capturing an image failed with error %d\n");

    /* Prepare image for extraction, may need to be rescaled. */
    const uint16_t rows = (*image)->rows;
    const uint16_t cols = (*image)->cols;
    const uint16_t resolution = (*image)->resolution;
    uint16_t extraction_rows = 0;
    uint16_t extraction_cols = 0;
    status = pb_embedded_compute_extraction_image_size (embedded, rows, cols, resolution,
                                                        &extraction_rows, &extraction_cols);
    CHECK_STATUS (status, "Computing extraction size failed with error %d\n");
    if (extraction_rows != rows) {
        /* Image needs to be rescaled. */
        pb_embedded_image_t* rescaled_image = pb_embedded_image_create (NULL, extraction_rows, extraction_cols, resolution, NULL);

        /* Rescale image. */
        status = pb_embedded_scale_image_for_extraction (embedded, *image, rescaled_image);
        CHECK_STATUS (status, "Scaling image for extraction size failed with error %d\n");
        pb_embedded_image_delete (*image);
        *image = rescaled_image;
    }

    return status;
}

static pb_rc_t
enroll (int finger_id)
{
    pb_rc_t status = PB_RC_OK;
    int nbr_of_captures = 0;

    /* Create embedded object. */
    pb_embedded_t* embedded = NULL;
    status = create_embedded (&embedded);
    CHECK_STATUS (status, "Creating embedded object failed with error %d\n");

    /* Load and decode the SPD object. */
    pb_spd_t* spd = NULL;
    status = load_spd (&spd);
    if (status != PB_RC_OK) return status;

    /* Perform enrollment. */
    printf ("Enrolling finger id %d:\n", finger_id);

    /* Initialize enrollment. */
    pb_storage_finger_t* finger = pb_flash_storage_get_finger (FlashSubtemplateStorage, finger_id);
    if (finger == NULL) printf ("finger_id not valid");
    status = pb_embedded_enroll_begin (embedded, &pb_flash_storageI, finger);
    CHECK_STATUS (status, "Initializing enrollment failed with error %d\n");

    for (int i = 0; i < PB_EMBEDDED_EXAMPLE_MAX_ENROLLMENT_CAPTURES; i++) {
        /* Capture an image from the sensor. */
        pb_embedded_image_t* image = NULL;
        status = capture_image (embedded, &image);
        nbr_of_captures++;

        /* Screen image. */
        pb_embedded_extract_screen_settings_t extract_screen_settings = pb_embedded_get_default_extract_screen_settings ();
        pb_embedded_image_quality_t quality = { 0 };
        status = pb_embedded_extract_screen (embedded, image, &extract_screen_settings, &quality);
        CHECK_STATUS (status, "Image screening failed with error %d\n");

        /* Run static pattern detection. Will set mask in image. */
        pb_embedded_spd_settings_t spd_settings = pb_embedded_spd_get_default_settings ();
        status = pb_embedded_spd_detect (embedded, image, &spd_settings, spd);
        CHECK_STATUS (status, "Static pattern detection failed with error %d\n");

        /* Extract template from image. */
        pb_template_t* template_ = NULL;
        status = pb_embedded_extract (embedded, image, &template_);
        CHECK_STATUS (status, "Extraction failed with error %d\n");

        /* Delete image. */
        pb_embedded_image_delete (image);

        /* Screen template. */
        pb_embedded_enroll_geometry_t geometry = { 0 };
        pb_embedded_enroll_screen_settings_t enroll_screen_settings = pb_embedded_get_default_enroll_screen_settings ();
        const int nbr_of_subtemplates = pb_storage_finger_get_nbr_of_subtemplates (finger);
        if (nbr_of_subtemplates > 0) { /* Only screen if there is something to compare to. */
            status = pb_embedded_enroll_screen (embedded, template_, &enroll_screen_settings, &geometry);
            CHECK_STATUS (status, "Template screening failed with error %d\n");
        }

        /* Add template. */
        uint8_t nbr_of_enrolled_subtemplates = nbr_of_subtemplates;
        if (external_enrollment_image_shall_be_added (&quality, &geometry)) {
            status = pb_embedded_enroll_add (embedded, template_);
            CHECK_STATUS (status, "Adding template failed with error %d\n");
            nbr_of_enrolled_subtemplates = pb_storage_finger_get_nbr_of_subtemplates (finger);
            printf ("Image %2d enrolled: nbr_of_enrolled_subtemplates=%2d\n", i, nbr_of_enrolled_subtemplates);
        }

        /* Delete template. */
        pb_template_delete (template_);
        template_ = NULL;

        /* Check if the enrollment is complete. */
        if (external_enrollment_complete (nbr_of_captures, nbr_of_enrolled_subtemplates)) {
            status = pb_embedded_enroll_end (embedded);
            CHECK_STATUS (status, "Finalizing enrollment failed with error %d.\n");
            break;
        }

    }

    /* Delete embedded object. */
    pb_embedded_delete (embedded);
    embedded = NULL;

    /* Encode and save the SPD object. */
    status = save_spd (spd);
    pb_embedded_spd_delete (spd);
    spd = NULL;

    return status;
}

static pb_rc_t
verify ()
{
    pb_rc_t status = PB_RC_OK;

    /* Create embedded object. */
    pb_embedded_t* embedded = NULL;
    status = create_embedded (&embedded);
    CHECK_STATUS (status, "Creating embedded object failed with error %d\n");

    /* Load and decode the SPD object. */
    pb_spd_t* spd = NULL;
    status = load_spd (&spd);
    if (status != PB_RC_OK) return status;

    pb_embedded_image_t* image = NULL;
    status = capture_image (embedded, &image);

    /* Screen image. */
    pb_embedded_extract_screen_settings_t extract_screen_settings = pb_embedded_get_default_extract_screen_settings ();
    pb_embedded_image_quality_t quality = { 0 };
    status = pb_embedded_extract_screen (embedded, image, &extract_screen_settings, &quality);
    CHECK_STATUS (status, "Image screening failed with error %d\n");

    if (external_use_image_for_verification (&quality)) {
        /* Image quality is ok. */

        /* Run static pattern detection. Will set mask in image. */
        pb_embedded_spd_settings_t spd_settings = pb_embedded_spd_get_default_settings ();
        status = pb_embedded_spd_detect (embedded, image, &spd_settings, spd);
        CHECK_STATUS (status, "Static pattern detection failed with error %d\n");

        /* Extract template from image. */
        pb_template_t* template_ = NULL;
        status = pb_embedded_extract (embedded, image, &template_);
        CHECK_STATUS (status, "Extraction failed with error %d\n");

        /* Retrieve enrolled fingers contexts. */
        const int nbr_of_enrolled_fingers = pb_flash_storage_get_nbr_of_enrolled_fingers (FlashSubtemplateStorage);
        pb_storage_finger_t** enrolled_fingers = malloc (nbr_of_enrolled_fingers * sizeof (pb_storage_finger_t*));
        status = pb_flash_storage_get_enrolled_fingers (FlashSubtemplateStorage, nbr_of_enrolled_fingers, enrolled_fingers);
        CHECK_STATUS (status, "Retrieving enrolled fingers failed with error %d\n");

        /* Perform the actual verification. */
        pb_embedded_verify_settings_t verify_settings = pb_embedded_get_default_verification_settings ();
        verify_settings.batch_size = MIN (PB_EMBEDDED_EXAMPLE_MAX_FINGER_BATCH_SIZE, nbr_of_enrolled_fingers);
        verify_settings.automatic_adjustment_of_far = 1;
        pb_match_result_t* match_result = NULL;
        status = pb_embedded_verify (embedded, &pb_flash_storageI, (void**)enrolled_fingers, nbr_of_enrolled_fingers,
                                     &verify_settings, template_, PB_EMBEDDED_EXAMPLE_FAR, &match_result);
        CHECK_STATUS (status, "Verification failed with error %d\n");

#ifdef PB_EMBEDDED_EXAMPLE_USE_LATENT
        const int decision_for_latent = pb_match_result_get_decision (match_result);
        /* Do latent detection if matching in verifier. */
        if (decision_for_latent == PB_DECISION_MATCH) {
            status = pb_latent_detection (embedded, image, match_result);
            CHECK_STATUS (status, "Latent detection failed with error %d\n");
        }
#endif
        /* Delete image. */
        pb_embedded_image_delete (image);

        /* Check result of verification. */
        const int decision = pb_match_result_get_decision (match_result);
        if (decision == PB_DECISION_MATCH) {
            const int matching_finger_index = pb_match_result_get_matching_template_index (match_result);
            pb_storage_finger_t* matching_finger = enrolled_fingers[matching_finger_index];
            /* Register accept so that storage may keep track of which fingers that was 
             * recently matched against to get optimal batch verification latency. */
            pb_flash_storage_register_accept (FlashSubtemplateStorage, matching_finger);
            printf ("Accept! Matched against finger id %d\n", pb_storage_finger_get_id (matching_finger));
        }
        else {
            printf ("Reject!\n");
        }

        /* Finalize verification.
         * In case of an accept the storage will be called to store the template. */
        status = pb_embedded_verify_finalize (embedded);
        CHECK_STATUS (status, "Finalizing verification failed with error %d\n");

        /* Delete match result. */
        pb_match_result_delete (match_result);
        match_result = NULL;

        /* Delete template. */
        pb_template_delete (template_);
        template_ = NULL;
    }
    else {
        /* Image rejected due to poor image quality. */
        printf ("Reject! Image quality too bad. \n");

        /* Delete image. */
        pb_embedded_image_delete (image);
    }

    /* Delete embedded object. */
    pb_embedded_delete (embedded);
    embedded = NULL;

    /* Encode and save the SPD object. */
    status = save_spd (spd);
    pb_embedded_spd_delete (spd);
    spd = NULL;

    return status;
}

pb_rc_t
terminate ()
{
    /* Terminate subtemplate storage. */
    pb_flash_storage_terminate (FlashSubtemplateStorage);
    FlashSubtemplateStorage = NULL;

#ifdef PB_EMBEDDED_EXAMPLE_USE_LATENT
    /* Terminate Latent storage. */
    pb_flash_data_storage_terminate (FlashLatentStorage);
    FlashLatentStorage = NULL;
#endif // PB_EMBEDDED_EXAMPLE_USE_LATENT

    /* Terminate SPD storage. */
    pb_flash_data_storage_terminate (FlashSPDStorage);
    FlashSPDStorage = NULL;

    /* Terminate FLASH implementation. */
    return external_flash_terminate (FlashContext);
}

int main (int argc, char* argv[])
{
    pb_rc_t status = PB_RC_OK;

    /* Init.
     * Normally performed only once at the start of the device's lifetime. */
    status = init ();
    if (status != PB_RC_OK) return -1;

    /* Enroll.
     * May be performed several times during the lifetime of the device. */
    for (int i = 0; i < PB_EMBEDDED_EXAMPLE_NBR_OF_FINGERS_TO_ENROLL; i++) {
        status = enroll (i);
        if (status != PB_RC_OK) return -1;
    }

    /* Verify.
     * Performed each time the user wants to access the device. */
    for (int i = 0; i < PB_EMBEDDED_EXAMPLE_NBR_OF_VERIFICATIONS; i++) {
        status = verify ();
        if (status != PB_RC_OK) return -1;
    }

    /* Terminate.
     * May never need to be called. */
    status = terminate ();
    if (status != PB_RC_OK) return -1;
    return 0;
}
